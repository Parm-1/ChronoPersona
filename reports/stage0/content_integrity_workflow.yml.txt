name: Content Integrity

on:
  push:
  pull_request:

permissions:
  contents: read

jobs:
  content-integrity:
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        python-version: ["3.11", "3.12"]

    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: ${{ matrix.python-version }}
          cache: pip

      - name: Install dependency-light project
        run: python -m pip install -e ".[dev]"

      - name: Validate structure without content access
        run: >-
          python scripts/validate_content_manifest.py
          --manifest tests/fixtures/content-integrity/manifest.jsonl
          --content-root tests/fixtures/content-integrity/documents

      - name: Validate content identities
        run: >-
          python scripts/validate_content_manifest.py
          --manifest tests/fixtures/content-integrity/manifest.jsonl
          --content-root tests/fixtures/content-integrity/documents
          --execute
          --output "${RUNNER_TEMP}/content-validation.json"

      - name: Plan audit without content access
        run: >-
          python scripts/audit_content_integrity.py
          --manifest tests/fixtures/content-integrity/manifest.jsonl
          --content-root tests/fixtures/content-integrity/documents
          --config configs/content-integrity-v0.json

      - name: Execute audit twice and compare exact output
        shell: bash
        run: |
          python scripts/audit_content_integrity.py \
            --manifest tests/fixtures/content-integrity/manifest.jsonl \
            --content-root tests/fixtures/content-integrity/documents \
            --config configs/content-integrity-v0.json \
            --execute \
            --output "${RUNNER_TEMP}/content-audit-one.json"
          python scripts/audit_content_integrity.py \
            --manifest tests/fixtures/content-integrity/manifest.jsonl \
            --content-root tests/fixtures/content-integrity/documents \
            --config configs/content-integrity-v0.json \
            --execute \
            --output "${RUNNER_TEMP}/content-audit-two.json"
          cmp \
            "${RUNNER_TEMP}/content-audit-one.json" \
            "${RUNNER_TEMP}/content-audit-two.json"

      - name: Assert fixture channels and scientific boundaries
        shell: bash
        run: |
          python - <<'PY'
          import json
          import os
          from pathlib import Path

          report = json.loads(
              (Path(os.environ["RUNNER_TEMP"]) / "content-audit-one.json")
              .read_text(encoding="utf-8")
          )
          assert report["summary"]["exact_raw_cluster_count"] == 1
          assert report["summary"]["exact_normalized_cluster_count"] == 2
          assert report["summary"]["near_duplicate_pair_count"] == 2
          assert report["summary"]["holdout_boundary_near_pair_count"] == 1
          assert report["summary"]["evaluation_exposure_pair_count"] == 1
          assert report["summary"]["direct_exposure_record_count"] == 1
          assert report["report_text_excerpts"] is False
          assert report["semantic_similarity_performed"] is False
          assert report["automatic_exclusion_performed"] is False
          PY

      - name: Run focused tests
        run: >-
          pytest -q
          tests/test_content_manifest.py
          tests/test_content_integrity.py
          tests/test_content_integrity_scripts.py
